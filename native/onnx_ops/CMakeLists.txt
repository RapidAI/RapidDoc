cmake_minimum_required(VERSION 3.24)
project(onnxruntime_ops VERSION 0.2.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75 80 86 89)
endif()
find_package(CUDAToolkit REQUIRED)
set(CUDNN_ROOT "" CACHE PATH "cuDNN9 directory containing include and lib")
set(CUDNN_FRONTEND_INCLUDE_DIR "" CACHE PATH "cuDNN frontend include directory")
set(NLOHMANN_JSON_INCLUDE_DIR "" CACHE PATH "directory containing nlohmann/json.hpp")
find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS "${CUDNN_ROOT}/include" REQUIRED)
find_library(CUDNN_LIBRARY NAMES cudnn libcudnn.so.9 HINTS "${CUDNN_ROOT}/lib" "${CUDNN_ROOT}/lib64" REQUIRED)
if(NOT EXISTS "${CUDNN_FRONTEND_INCLUDE_DIR}/cudnn_frontend.h")
  message(FATAL_ERROR "Set CUDNN_FRONTEND_INCLUDE_DIR (tested frontend v1.12.0)")
endif()
if(NOT EXISTS "${NLOHMANN_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set NLOHMANN_JSON_INCLUDE_DIR")
endif()
add_library(onnxruntime_ops SHARED src/ops.cc src/kernels.cu)
if(WIN32)
  target_compile_definitions(onnxruntime_ops PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(onnxruntime_ops PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
endif()
target_include_directories(onnxruntime_ops PRIVATE third_party/onnxruntime/include src "${CUDNN_INCLUDE_DIR}" "${CUDNN_FRONTEND_INCLUDE_DIR}" "${NLOHMANN_JSON_INCLUDE_DIR}")
target_link_libraries(onnxruntime_ops PRIVATE CUDA::cudart "${CUDNN_LIBRARY}")
set_target_properties(onnxruntime_ops PROPERTIES CUDA_SEPARABLE_COMPILATION OFF BUILD_RPATH "${CUDNN_ROOT}/lib;${CUDNN_ROOT}/lib64" INSTALL_RPATH "$ORIGIN")
install(TARGETS onnxruntime_ops LIBRARY DESTINATION onnx_ops/lib RUNTIME DESTINATION onnx_ops/lib)
